cmake_minimum_required(VERSION 3.16)
project(pangu_acl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CANN_ROOT "/usr/local/Ascend/cann" CACHE PATH "Qualified CANN installation")
find_path(ACL_INCLUDE acl/acl.h PATHS "${CANN_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${CANN_ROOT}/lib64" REQUIRED NO_DEFAULT_PATH)
find_library(OPAPI_LIBRARY opapi PATHS "${CANN_ROOT}/lib64" REQUIRED NO_DEFAULT_PATH)
find_library(NNOPBASE_LIBRARY nnopbase PATHS "${CANN_ROOT}/lib64" REQUIRED NO_DEFAULT_PATH)
find_library(HCCL_LIBRARY hccl PATHS "${CANN_ROOT}/lib64" REQUIRED NO_DEFAULT_PATH)
add_library(pangu_acl SHARED src/pangu_acl.cpp)
target_include_directories(pangu_acl PUBLIC include PRIVATE "${ACL_INCLUDE}")
target_link_libraries(pangu_acl PRIVATE "${HCCL_LIBRARY}" "${ACL_LIBRARY}" "${OPAPI_LIBRARY}" "${NNOPBASE_LIBRARY}")
target_compile_options(pangu_acl PRIVATE -Wall -Wextra -Werror)
add_executable(acl_graph_probe probes/acl_graph_probe.cpp)
target_link_libraries(acl_graph_probe PRIVATE pangu_acl)
add_executable(gdr_contract_probe probes/gdr_contract_probe.cpp)
target_include_directories(gdr_contract_probe PRIVATE "${ACL_INCLUDE}")
target_link_libraries(gdr_contract_probe PRIVATE "${ACL_LIBRARY}" "${OPAPI_LIBRARY}" "${NNOPBASE_LIBRARY}")
target_compile_options(gdr_contract_probe PRIVATE -Wall -Wextra -Werror)

add_executable(sampler_probe probes/sampler_probe.cpp)
target_link_libraries(sampler_probe PRIVATE pangu_acl)
target_compile_options(sampler_probe PRIVATE -Wall -Wextra -Werror)

add_executable(tp_probe probes/tp_probe.cpp)
target_link_libraries(tp_probe PRIVATE pangu_acl pthread)
target_compile_options(tp_probe PRIVATE -Wall -Wextra -Werror)
